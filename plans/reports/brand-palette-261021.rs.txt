//! The light and dark palettes, and the timeline's track colours.

use gpui::{Rgba, rgb, rgba};

use crate::{Appearance, EditorPalette, Theme, mix};

/// Opaque colour from `0xRRGGBB`.
pub const fn hex(value: u32) -> Rgba {
    Rgba {
        r: ((value >> 16) & 0xff) as f32 / 255.,
        g: ((value >> 8) & 0xff) as f32 / 255.,
        b: (value & 0xff) as f32 / 255.,
        a: 1.,
    }
}

/// Colour with alpha from `0xRRGGBBAA`.
pub const fn hexa(value: u32) -> Rgba {
    Rgba {
        r: ((value >> 24) & 0xff) as f32 / 255.,
        g: ((value >> 16) & 0xff) as f32 / 255.,
        b: ((value >> 8) & 0xff) as f32 / 255.,
        a: (value & 0xff) as f32 / 255.,
    }
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum TrackKind {
    Clip,
    Overlay,
    Audio,
}

/// Muted base colours: cobalt scenes, peach overlays, mint audio.
pub const fn track_color(kind: TrackKind) -> Rgba {
    match kind {
        TrackKind::Clip => hex(0x2864f5),
        TrackKind::Overlay => hex(0xcb8b53),
        TrackKind::Audio => hex(0x4b9c7a),
    }
}

#[derive(Debug, Clone, Copy, PartialEq)]
pub struct SegmentColors {
    pub fill: Rgba,
    pub border: Rgba,
    pub label: Rgba,
}

/// Timeline surfaces use the landing page's powder blue, peach and mint.
/// Dark mode tints navy panels with each track colour; selection deepens the fill.
pub fn segment_colors_for(
    appearance: Appearance,
    kind: TrackKind,
    selected: bool,
) -> SegmentColors {
    let palette = match appearance {
        Appearance::Light => EditorPalette::light(),
        Appearance::Dark => EditorPalette::dark(),
    };
    let track = track_color(kind);
    let opaque = |color: Rgba| Rgba { a: 1., ..color };
    let fill = match appearance {
        Appearance::Light => {
            let surface = match kind {
                TrackKind::Clip => hex(0xedf3ff),
                TrackKind::Overlay => hex(0xfff0e3),
                TrackKind::Audio => hex(0xe5f4ef),
            };
            mix(surface, track, if selected { 0.10 } else { 0. })
        }
        Appearance::Dark => mix(palette.card, track, if selected { 0.30 } else { 0.20 }),
    };
    SegmentColors {
        fill: opaque(fill),
        border: opaque(mix(
            palette.card,
            track,
            if appearance == Appearance::Light {
                0.34
            } else {
                0.42
            },
        )),
        label: opaque(mix(
            track,
            palette.text_1,
            if appearance == Appearance::Light {
                0.70
            } else {
                0.55
            },
        )),
    }
}

#[cfg(test)]
mod tests {
    use super::*;

    /// `0xRRGGBBAA`, rounded from the stored floats.
    fn bytes(color: Rgba) -> u32 {
        let byte = |channel: f32| (channel * 255.).round() as u32;
        byte(color.r) << 24 | byte(color.g) << 16 | byte(color.b) << 8 | byte(color.a)
    }

    fn assert_close(actual: Rgba, expected: u32, name: &str) {
        let (r, g, b) = (
            (expected >> 16) & 0xff,
            (expected >> 8) & 0xff,
            expected & 0xff,
        );
        let got = bytes(actual) >> 8;
        let delta = |a: u32, b: u32| a.abs_diff(b);
        assert!(
            delta((got >> 16) & 0xff, r) <= 1
                && delta((got >> 8) & 0xff, g) <= 1
                && delta(got & 0xff, b) <= 1,
            "{name}: got {got:06x}, want {expected:06x}"
        );
    }

    #[test]
    fn segment_colors_follow_mix_rule() {
        // Light scene fills match the landing page powder blue exactly.
        // Dark audio fills retain their navy tint and mint identity.
        let light = segment_colors_for(Appearance::Light, TrackKind::Clip, false);
        assert_close(light.fill, 0xedf3ff, "light fill");
        assert_close(light.border, 0xb6cafc, "light border");
        assert_close(light.label, 0x1b3b7e, "light label");
        let light_selected = segment_colors_for(Appearance::Light, TrackKind::Clip, true);
        assert_close(light_selected.fill, 0xd9e5fe, "light selected fill");
        assert_eq!(light_selected.border, light.border);

        let dark = segment_colors_for(Appearance::Dark, TrackKind::Audio, false);
        assert_close(dark.fill, 0x1d3742, "dark fill");
        assert_close(dark.border, 0x295351, "dark border");
        assert_close(dark.label, 0xabd1c3, "dark label");
        let dark_selected = segment_colors_for(Appearance::Dark, TrackKind::Audio, true);
        assert_close(dark_selected.fill, 0x224449, "dark selected fill");
        for color in [light.fill, light.border, light.label, dark.fill] {
            assert_eq!(color.a, 1.);
        }
    }
}
